# Install Arduino CLI
curl -fsSL https://raw.githubusercontent.com/arduino/arduino-cli/master/install.sh | sh

# Add to PATH
echo 'export PATH=$PATH:$HOME/bin' >> ~/.bashrc
source ~/.bashrc

# Install Arduino core
arduino-cli core install arduino:avr

# Upload the simulator
cd /Users/adrielrivera/Documents/SICK7/SICK-App
arduino-cli compile --fqbn arduino:avr:uno arduino_pbt_simulator.ino
arduino-cli upload -p /dev/ttyUSB0 --fqbn arduino:avr:uno arduino_pbt_simulator.ino
